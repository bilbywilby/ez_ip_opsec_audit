atomic_helper.py – creates a small repo, writes a file via a heredoc,
prepends a timestamp using `tac`, and commits it with Git.
All actions are performed in a single, self‑contained run.
"""

import os
import subprocess
import datetime
import textwrap
from pathlib import Path

# ------------------------------------------------------------------
# Helper to run a command atomically (raises on error)
def run(cmd: list, cwd: Path = None):
    subprocess.run(cmd, cwd=cwd, check=True, text=True, stdout=subprocess.PIPE, stderr=subprocess.PIPE)


# 1. Create a fresh directory
repo_dir = Path.home() / "atomic_repo"
repo_dir.mkdir(parents=True, exist_ok=True)

# 2. Initialise a git repo (quiet)
run(["git", "init", "--quiet"], cwd=repo_dir)

# 3. Write a file using a heredoc (via bash)
target_file = repo_dir / "log.txt"
heredoc_content = textwrap.dedent("""\
    ========================================
    Session started at: {timestamp}
    ========================================
    This is a generated log entry.
    """)

# Insert the actual timestamp
filled = heredoc_content.format(timestamp=datetime.datetime.utcnow().isoformat())

# Use bash heredoc to create the file atomically
run(["bash", "-c", f"cat > {target_file} <<'EOF'\n{filled}\nEOF"])

# 4. Prepend a new line with the current time using `tac`
new_line = f"# checkpoint {datetime.datetime.utcnow().isoformat()}\n"
tmp_file = repo_dir / ".tmp_log"

# Echo the new line, then append the existing file reversed twice (tac + tac = original order)
run(["bash", "-c",
     f"printf '{new_line}' > {tmp_file} && tac {target_file} >> {tmp_file} && mv {tmp_file} {target_file}"])

# 5. Add & commit with Git
run(["git", "add", "log.txt"], cwd=repo_dir)
run(["git", "commit", "-m", "Atomic log update"], cwd=repo_dir)

print(f"✅ Repo created at {repo_dir}\n"
      f"   log.txt →\n{Path(target_file).read_text()}")

4. **`tac`** – prepends a “checkpoint” line without loading the whole file into Python memory (`tac` reverses, we append, then move back).  
5. **Git add/commit** – stores the change.
